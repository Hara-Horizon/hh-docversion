9#[derive(Debug, Error)]
10pub enum ContractError {
11    #[error("SemVer parsing error: {0}")]
12    Semver(#[from] semver::Error),
13    #[error("Artifact I/O failure: {0}")]
14    Io(#[from] io::Error),
15    #[error("Contract violation: App v{app_ver} fails requirements '{range}' for schema '{schema_id}' v{schema_ver}")]
16    RangeMismatch {
17        app_ver: Version,
18        range: String,
19        schema_id: String,
20        schema_ver: u32,
21    },
22    #[error("Contract absent: Schema '{schema_id}' v{schema_ver} is not declared in manifest")]
23    NotFound { schema_id: String, schema_ver: u32 },
24    #[error("Integrity violation for '{path}': expected {expected}, calculated {actual}")]
25    ChecksumMismatch {
26        path: String,
27        expected: String,
28        actual: String,
29    },
30}

47struct Sha256Writer<D: Digest> {
48    hasher: D,
49}
50
51impl<D: Digest> Write for Sha256Writer<D> {
52    fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
53        self.hasher.update(buf);
54        Ok(buf.len())
55    }
56
57    fn flush(&mut self) -> io::Result<()> {
58        Ok(())
59    }
60}

76    /// Verifies whether the specified application version satisfies the schema contract requirement.
77    ///
78    /// ```rust
79    /// use docversion_manifest::{SchemaContract, SchemaMatrix};
80    ///
81    /// let matrix = SchemaMatrix {
82    ///     generated_at: "2026-11-07T00:00:00Z".into(),
83    ///     contracts: vec![SchemaContract {
84    ///         schema_id: "core-ledger".into(),
85    ///         schema_version: 4,
86    ///         supported_app_range: ">=1.4.0, <2.0.0".into(),
87    ///         artifact_uri: "[https://example.com/db.sqlite](https://example.com/db.sqlite)".into(),
88    ///         sha256: "abc".into(),
89    ///     }],
90    /// };
91    ///
92    /// assert!(matrix.verify_compatibility("1.5.0", "core-ledger", 4).is_ok());
93    /// assert!(matrix.verify_compatibility("2.1.0", "core-ledger", 4).is_err());
94    /// ```
95    pub fn verify_compatibility(
96        &self,
97        app_version_str: &str,
98        target_schema_id: &str,
99        actual_db_version: u32,
100    ) -> Result<&SchemaContract, ContractError> {
101        let app_ver = Version::parse(app_version_str.trim())?;
102
103        let contract = self
104            .contracts
105            .iter()
106            .find(|c| c.schema_id == target_schema_id && c.schema_version == actual_db_version)
107            .ok_or_else(|| ContractError::NotFound {
108                schema_id: target_schema_id.to_string(),
109                schema_ver: actual_db_version,
110            })?;
111
112        let req = VersionReq::parse(&contract.supported_app_range)?;
113        if !req.matches(&app_ver) {
114            return Err(ContractError::RangeMismatch {
115                app_ver,
116                range: contract.supported_app_range.clone(),
117                schema_id: target_schema_id.to_string(),
118                schema_ver: actual_db_version,
119            });
120        }
121
122        Ok(contract)
123    }
124
125    /// Computes the artifact digest using a zero-allocation buffered streaming pipeline.
126    pub fn verify_file_checksum<P: AsRef<Path>>(
127        file_path: P,
128        expected_sha256: &str,
129    ) -> Result<(), ContractError> {
130        let file = File::open(file_path.as_ref())?;
131        let mut reader = BufReader::with_capacity(64 * 1024, file);
132        let mut writer = Sha256Writer { hasher: Sha256::new() };
133
134        io::copy(&mut reader, &mut writer)?;
135        let actual_hash = format!("{:x}", writer.hasher.finalize());
136
137        if !actual_hash.eq_ignore_ascii_case(expected_sha256.trim()) {
138            return Err(ContractError::ChecksumMismatch {
139                path: file_path.as_ref().display().to_string(),
140                expected: expected_sha256.to_string(),
141                actual: actual_hash,
142            });
143        }
144
145        Ok(())
146    }
147}

155    #[test]
156    fn test_matrix_contract_verification() {
157        let matrix = SchemaMatrix {
158            generated_at: "2026-11-07T00:00:00Z".into(),
159            contracts: vec![SchemaContract {
160                schema_id: "core-ledger".into(),
161                schema_version: 4,
162                supported_app_range: ">=1.4.0, <2.0.0".into(),
163                artifact_uri: "https://example.com/db.sqlite".into(),
164                sha256: "abc".into(),
165            }],
166        };
167
168        assert!(matrix.verify_compatibility("1.5.0", "core-ledger", 4).is_ok());
169        assert!(matrix.verify_compatibility("2.1.0", "core-ledger", 4).is_err());
170        assert!(matrix.verify_compatibility("1.5.0", "core-ledger", 5).is_err());
171    }

174    fn test_streaming_checksum_verification() {
175        let nanos = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_nanos();
176        let tmp_path = std::env::temp_dir().join(format!("docversion_stream_test_{}.bin", nanos));

178        {

181            file.flush().unwrap();

184        let expected = "ba93a56a8fe9befface03cc6b575aebef8cb8cde4d259452aeea11cab154fb74";

186        assert!(SchemaMatrix::verify_file_checksum(&tmp_path, "deadbeef").is_err());
