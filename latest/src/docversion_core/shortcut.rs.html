1//! Clean Markdown exporter and rich-text HTML converter for Shortcut Docs compatibility.
2//!
3//! Strips raw HTML tags outside fenced code blocks, normalizes internal
4//! Rustdoc/site links to clean relative Markdown paths, and converts
5//! sanitized Markdown (including GFM pipe tables and fenced code blocks)
6//! to semantic HTML with language-agnostic syntax highlighting for RTF paste.

15/// Sanitizes generated Markdown for zero-friction import into Shortcut Docs.
16///
17/// - Preserves fenced code blocks (` ``` ` and `~~~`) and inline backtick spans verbatim.
18/// - Converts `<br>` / `<br/>` into Markdown hard line breaks (`  \n`).
19/// - Converts `<details>` / `<summary>` blocks into clean Markdown bold headers.
20/// - Strips all other raw HTML tags (`<div>`, `<span>`, `<a name="...">`, etc.) while keeping inner text.
21/// - Rewrites `.html` and root-prefixed (`/latest/...`, `/vX.Y.Z/...`) links to relative `.md` paths.
22pub fn sanitize_for_shortcut_docs(markdown: &str) -> String {
23    let mut output = String::with_capacity(markdown.len());
24    let mut in_fence = false;
25    let mut fence_marker: Option<&str> = None;
26
27    for line in markdown.split_inclusive('\n') {
28        let trimmed = line.trim_start();
29
30        if let Some(marker) = detect_fence_prefix(trimmed) {
31            if !in_fence {
32                in_fence = true;
33                fence_marker = Some(marker);
34            } else if fence_marker == Some(marker) {
35                in_fence = false;
36                fence_marker = None;
37            }
38            output.push_str(line);
39            continue;
40        }
41
42        if in_fence {
43            output.push_str(line);
44            continue;
45        }
46
47        let processed = process_prose_line(line);
48        output.push_str(&processed);
49    }
50
51    collapse_excess_blank_lines(&output)
52}

54/// Converts sanitized Shortcut Markdown into compact, well-spaced semantic HTML with
55/// GFM table support and language-agnostic syntax highlighting for RTF conversion.
56pub fn markdown_to_clean_html(markdown: &str) -> String {
57    let mut html = String::with_capacity(markdown.len() * 2 + 1024);
58    html.push_str(
59        r#"<!DOCTYPE html>
60<html>
61<head>
62<meta charset="utf-8">
63<style>
64  body {
65    font-family: -apple-system, BlinkMacSystemFont, "SF Pro Text", "Helvetica Neue", sans-serif;
66    font-size: 14px;
67    line-height: 1.55;
68    color: #1f2328;
69  }
70  h1 {
71    font-size: 24px;
72    font-weight: 700;
73    color: #0f141c;
74  }
75  h2 {
76    font-size: 19px;
77    font-weight: 600;
78    color: #0f141c;
79  }
80  h3 {
81    font-size: 16px;
82    font-weight: 600;
83    color: #1f2328;
84  }
85  h4, h5, h6 {
86    font-size: 14px;
87    font-weight: 600;
88    color: #424a53;
89  }
90  blockquote {
91    padding: 8px 14px;
92    border-left: 3px solid #d97706;
93    background-color: #f6f8fa;
94    color: #424a53;
95  }
96  li {
97    line-height: 1.55;
98  }
99  .spacer {
100    font-size: 6px;
101    line-height: 6px;
102  }
103</style>
104</head>
105<body>
106"#,
107    );
108
109    let spacer = "<p class=\"spacer\">&nbsp;</p>\n";
110    let lines: Vec<&str> = markdown.lines().collect();
111    let mut idx = 0;
112    let mut in_fence = false;
113    let mut fence_marker: Option<&str> = None;
114    let mut in_list = false;
115    let mut has_emitted_block = false;
116    let mut last_was_spacer = false;
117
118    while idx < lines.len() {
119        let line = lines[idx];
120        let trimmed = line.trim_start();
121
122        if let Some(marker) = detect_fence_prefix(trimmed) {
123            if in_list {
124                html.push_str("</ul>\n");
125                in_list = false;
126            }
127            if !in_fence {
128                in_fence = true;
129                fence_marker = Some(marker);
130                html.push_str(&format!("<pre style=\"{PRE_STYLE}\"><code>"));
131            } else if fence_marker == Some(marker) {
132                in_fence = false;
133                fence_marker = None;
134                html.push_str("</code></pre>\n");
135                has_emitted_block = true;
136                last_was_spacer = false;
137            } else {
138                html.push_str(&highlight_code_universal(line));
139                html.push('\n');
140            }
141            idx += 1;
142            continue;
143        }
144
145        if in_fence {
146            html.push_str(&highlight_code_universal(line));
147            html.push('\n');
148            idx += 1;
149            continue;
150        }
151
152        if trimmed.is_empty() {
153            if in_list {
154                html.push_str("</ul>\n");
155                in_list = false;
156            }
157            idx += 1;
158            continue;
159        }
160
161        // GFM Pipe Table (`| H1 | H2 |` followed by `| --- | --- |`)
162        if is_pipe_table_row(trimmed)
163            && idx + 1 < lines.len()
164            && is_pipe_table_separator(lines[idx + 1].trim())
165        {
166            if in_list {
167                html.push_str("</ul>\n");
168                in_list = false;
169            }
170            let headers = parse_pipe_cells(trimmed);
171            html.push_str(&format!("<table style=\"{TABLE_STYLE}\">\n<thead><tr>"));
172            for h in headers {
173                let cell_html = render_inline_html(&h);
174                html.push_str(&format!("<th style=\"{TH_STYLE}\">{cell_html}</th>"));
175            }
176            html.push_str("</tr></thead>\n<tbody>\n");
177            idx += 2; // skip header + separator
178
179            while idx < lines.len() && is_pipe_table_row(lines[idx].trim()) {
180                let cells = parse_pipe_cells(lines[idx].trim());
181                html.push_str("<tr>");
182                for c in cells {
183                    let cell_html = render_inline_html(&c);
184                    html.push_str(&format!("<td style=\"{TD_STYLE}\">{cell_html}</td>"));
185                }
186                html.push_str("</tr>\n");
187                idx += 1;
188            }
189            html.push_str("</tbody>\n</table>\n");
190            has_emitted_block = true;
191            last_was_spacer = false;
192            continue;
193        }
194
195        if trimmed == "---" || trimmed == "***" || trimmed == "___" {
196            if in_list {
197                html.push_str("</ul>\n");
198                in_list = false;
199            }
200            html.push_str("<hr/>\n");
201            has_emitted_block = true;
202            last_was_spacer = false;
203            idx += 1;
204            continue;
205        }
206
207        // Headings (# through ######): emit at most one spacer immediately before h1-h3
208        let hash_count = trimmed.bytes().take_while(|&b| b == b'#').count();
209        if (1..=6).contains(&hash_count) && trimmed.as_bytes().get(hash_count) == Some(&b' ') {
210            if in_list {
211                html.push_str("</ul>\n");
212                in_list = false;
213            }
214            if has_emitted_block && hash_count <= 3 && !last_was_spacer {
215                html.push_str(spacer);
216            }
217            let text = render_inline_html(trimmed[hash_count + 1..].trim());
218            html.push_str(&format!("<h{hash_count}>{text}</h{hash_count}>\n"));
219            has_emitted_block = true;
220            last_was_spacer = false;
221            idx += 1;
222            continue;
223        }
224
225        // Blockquotes (> ...)
226        if let Some(quote_body) = trimmed.strip_prefix("> ") {
227            if in_list {
228                html.push_str("</ul>\n");
229                in_list = false;
230            }
231            let text = render_inline_html(quote_body);
232            html.push_str(&format!("<blockquote><p>{text}</p></blockquote>\n"));
233            has_emitted_block = true;
234            last_was_spacer = false;
235            idx += 1;
236            continue;
237        }
238
239        // Unordered list items (* ... or - ...)
240        if let Some(item_body) = trimmed
241            .strip_prefix("* ")
242            .or_else(|| trimmed.strip_prefix("- "))
243        {
244            if !in_list {
245                html.push_str("<ul>\n");
246                in_list = true;
247            }
248            let text = render_inline_html(item_body);
249            html.push_str(&format!("<li>{text}</li>\n"));
250            has_emitted_block = true;
251            last_was_spacer = false;
252            idx += 1;
253            continue;
254        }
255
256        if in_list {
257            html.push_str("</ul>\n");
258            in_list = false;
259        }
260
261        let text = render_inline_html(trimmed);
262        html.push_str(&format!("<p>{text}</p>\n"));
263        has_emitted_block = true;
264        last_was_spacer = false;
265        idx += 1;
266    }
267
268    if in_list {
269        html.push_str("</ul>\n");
270    }
271    if in_fence {
272        html.push_str("</code></pre>\n");
273    }
274
275    html.push_str("</body></html>\n");
276    html
277}

279fn is_pipe_table_row(line: &str) -> bool {
280    let trimmed = line.trim();
281    trimmed.starts_with('|') && trimmed.ends_with('|') && trimmed.len() >= 3
282}

284fn is_pipe_table_separator(line: &str) -> bool {
285    if !is_pipe_table_row(line) {
286        return false;
287    }
288    let cells = parse_pipe_cells(line);
289    !cells.is_empty()
290        && cells.iter().all(|c| {
291            let s = c.trim();
292            !s.is_empty() && s.chars().all(|ch| ch == '-' || ch == ':')
293        })
294}

296fn parse_pipe_cells(line: &str) -> Vec<String> {
297    let inner = line.trim().trim_start_matches('|').trim_end_matches('|');
298    inner.split('|').map(|c| c.trim().to_string()).collect()
299}

312/// Language-agnostic lexical syntax highlighter emitting uniform styled spans
313/// so macOS `textutil` RTF conversion preserves consistent weight and monospace runs.
314pub fn highlight_code_universal(line: &str) -> String {
315    let trimmed = line.trim_start();
316    if trimmed.starts_with("//") || trimmed.starts_with('#') || trimmed.starts_with("--") {
317        return span(CODE_CMT_STYLE, line);
318    }
319
320    let mut out = String::with_capacity(line.len() * 3);
321    let chars: Vec<char> = line.chars().collect();
322    let len = chars.len();
323    let mut i = 0;
324
325    while i < len {
326        // Inline // comment
327        if chars[i] == '/' && i + 1 < len && chars[i + 1] == '/' {
328            let rest: String = chars[i..].iter().collect();
329            out.push_str(&span(CODE_CMT_STYLE, &rest));
330            break;
331        }
332
333        // String literal ("..." or '...')
334        if chars[i] == '"' || chars[i] == '\'' {
335            let quote = chars[i];
336            let start = i;
337            i += 1;
338            while i < len {
339                if chars[i] == '\\' && i + 1 < len {
340                    i += 2;
341                    continue;
342                }
343                if chars[i] == quote {
344                    i += 1;
345                    break;
346                }
347                i += 1;
348            }
349            let lit: String = chars[start..i].iter().collect();
350            out.push_str(&span(CODE_STR_STYLE, &lit));
351            continue;
352        }
353
354        // Numeric literal
355        if chars[i].is_ascii_digit() && (i == 0 || !chars[i - 1].is_ascii_alphanumeric()) {
356            let start = i;
357            while i < len
358                && (chars[i].is_ascii_alphanumeric() || chars[i] == '.' || chars[i] == '_')
359            {
360                i += 1;
361            }
362            let num: String = chars[start..i].iter().collect();
363            out.push_str(&span(CODE_NUM_STYLE, &num));
364            continue;
365        }
366
367        // Identifier / keyword / PascalCase type
368        if chars[i].is_ascii_alphabetic() || chars[i] == '_' {
369            let start = i;
370            while i < len && (chars[i].is_ascii_alphanumeric() || chars[i] == '_') {
371                i += 1;
372            }
373            let ident: String = chars[start..i].iter().collect();
374            if is_universal_keyword(&ident) {
375                out.push_str(&span(CODE_KW_STYLE, &ident));
376            } else if ident.chars().next().is_some_and(|c| c.is_ascii_uppercase())
377                && ident.chars().any(|c| c.is_ascii_lowercase())
378            {
379                out.push_str(&span(CODE_TYPE_STYLE, &ident));
380            } else {
381                out.push_str(&span(CODE_PLAIN_STYLE, &ident));
382            }
383            continue;
384        }
385
386        // Group contiguous whitespace/punctuation into a single uniform plain span
387        let start = i;
388        while i < len
389            && !chars[i].is_ascii_alphanumeric()
390            && chars[i] != '_'
391            && chars[i] != '"'
392            && chars[i] != '\''
393            && !(chars[i] == '/' && i + 1 < len && chars[i + 1] == '/')
394        {
395            i += 1;
396        }
397        let punct: String = chars[start..i].iter().collect();
398        out.push_str(&span(CODE_PLAIN_STYLE, &punct));
399    }
400
401    out
402}

472fn render_inline_html(line: &str) -> String {
473    let mut out = String::with_capacity(line.len());
474    let mut in_code = false;
475    let mut buf = String::new();
476
477    for ch in line.chars() {
478        if ch == '`' {
479            if in_code {
480                out.push_str(&format!("<code style=\"{INLINE_CODE_STYLE}\">"));
481                out.push_str(&escape_html(&buf));
482                out.push_str("</code>");
483                buf.clear();
484                in_code = false;
485            } else {
486                out.push_str(&apply_bold_html(&escape_html(&buf)));
487                buf.clear();
488                in_code = true;
489            }
490        } else {
491            buf.push(ch);
492        }
493    }
494
495    if !buf.is_empty() {
496        if in_code {
497            out.push_str(&format!("<code style=\"{INLINE_CODE_STYLE}\">"));
498            out.push_str(&escape_html(&buf));
499            out.push_str("</code>");
500        } else {
501            out.push_str(&apply_bold_html(&escape_html(&buf)));
502        }
503    }
504
505    out
506}

508fn apply_bold_html(escaped: &str) -> String {
509    let mut out = String::with_capacity(escaped.len());
510    let mut rest = escaped;
511    while let Some(start) = rest.find("**") {
512        out.push_str(&rest[..start]);
513        let after_start = &rest[start + 2..];
514        if let Some(end) = after_start.find("**") {
515            out.push_str("<strong>");
516            out.push_str(&after_start[..end]);
517            out.push_str("</strong>");
518            rest = &after_start[end + 2..];
519        } else {
520            out.push_str("**");
521            rest = after_start;
522            break;
523        }
524    }
525    out.push_str(rest);
526    out
527}

539/// Processes a single non-fenced Markdown line, respecting inline code spans (`...`).
540fn process_prose_line(line: &str) -> String {
541    let mut result = String::with_capacity(line.len());
542    let mut chars = line.char_indices().peekable();
543    let mut in_inline_code = false;
544    let mut inline_ticks = 0usize;
545    let mut prose_buf = String::new();
546
547    while let Some((_idx, ch)) = chars.next() {
548        if ch == '`' {
549            let mut tick_count = 1;
550            while let Some(&(_, '`')) = chars.peek() {
551                chars.next();
552                tick_count += 1;
553            }
554
555            if !in_inline_code {
556                if !prose_buf.is_empty() {
557                    result.push_str(&transform_prose_segment(&prose_buf));
558                    prose_buf.clear();
559                }
560                in_inline_code = true;
561                inline_ticks = tick_count;
562            } else if tick_count == inline_ticks {
563                in_inline_code = false;
564                inline_ticks = 0;
565            }
566
567            for _ in 0..tick_count {
568                result.push('`');
569            }
570        } else if in_inline_code {
571            result.push(ch);
572        } else {
573            prose_buf.push(ch);
574        }
575    }
576
577    if !prose_buf.is_empty() {
578        result.push_str(&transform_prose_segment(&prose_buf));
579    }
580
581    result
582}

590/// Strips raw HTML tags while converting structural tags (`<br>`, `<summary>`) to Markdown.
591fn strip_html_tags(input: &str) -> String {
592    let mut out = String::with_capacity(input.len());
593    let bytes = input.as_bytes();
594    let mut i = 0;
595
596    while i < bytes.len() {
597        if bytes[i] == b'<' {
598            if let Some(&next) = bytes.get(i + 1) {
599                let is_tag_start =
600                    next == b'/' || next == b'!' || (next as char).is_ascii_alphabetic();
601
602                if is_tag_start {
603                    if let Some(close_offset) = input[i..].find('>') {
604                        let raw_tag = &input[i..=i + close_offset];
605                        let replacement = map_structural_tag(raw_tag);
606                        out.push_str(replacement);
607                        i += close_offset + 1;
608                        continue;
609                    }
610                }
611            }
612        }
613
614        let ch = input[i..].chars().next().unwrap();
615        out.push(ch);
616        i += ch.len_utf8();
617    }
618
619    out
620}

622/// Maps select structural HTML tags to clean Markdown equivalents; drops all others.
623fn map_structural_tag(raw_tag: &str) -> &'static str {
624    let inner = raw_tag.trim_start_matches('<').trim_end_matches('>').trim();
625
626    let tag_name = inner
627        .trim_start_matches('/')
628        .split(|c: char| c.is_whitespace() || c == '/')
629        .next()
630        .unwrap_or("")
631        .to_ascii_lowercase();
632
633    let is_closing = inner.starts_with('/');
634
635    match tag_name.as_str() {
636        "br" => "  \n",
637        "hr" => "\n---\n",
638        "summary" if !is_closing => "**",
639        "summary" if is_closing => "**\n",
640        _ => "",
641    }
642}

644/// Rewrites `[label](target)` Markdown links to clean relative `.md` paths for Shortcut Docs.
645fn rewrite_markdown_links(input: &str) -> String {
646    let mut out = String::with_capacity(input.len());
647    let mut rest = input;
648
649    while let Some(open_bracket) = rest.find('[') {
650        out.push_str(&rest[..open_bracket]);
651        let after_open = &rest[open_bracket..];
652
653        if let Some(mid) = after_open.find("](") {
654            let label = &after_open[1..mid];
655            let after_mid = &after_open[mid + 2..];
656
657            if let Some(close_paren) = after_mid.find(')') {
658                let raw_target = after_mid[..close_paren].trim();
659                let clean_target = normalize_target_path(raw_target);
660
661                out.push('[');
662                out.push_str(label);
663                out.push_str("](");
664                out.push_str(&clean_target);
665                out.push(')');
666
667                rest = &after_mid[close_paren + 1..];
668                continue;
669            }
670        }
671
672        out.push('[');
673        rest = &after_open[1..];
674    }
675
676    out.push_str(rest);
677    out
678}

680/// Normalizes internal doc targets into relative `.md` paths while preserving external URLs.
681fn normalize_target_path(target: &str) -> String {
682    if target.starts_with("http://")
683        || target.starts_with("https://")
684        || target.starts_with("mailto:")
685    {
686        return target.to_string();
687    }
688
689    if let Some(anchor) = target.strip_prefix('#') {
690        return format!("#{}", slugify_anchor(anchor));
691    }
692
693    let (raw_path, fragment) = match target.split_once('#') {
694        Some((p, f)) => (p, Some(f)),
695        None => (target, None),
696    };
697
698    let stripped_prefix = strip_version_prefix(raw_path.trim_start_matches('/'));
699
700    let relative_md = if stripped_prefix.ends_with("/index.html") {
701        let dir = stripped_prefix.trim_end_matches("/index.html");
702        format!("./{dir}.md")
703    } else if stripped_prefix == "index.html" {
704        "./index.md".to_string()
705    } else if let Some(stem) = stripped_prefix.strip_suffix(".html") {
706        if stem.starts_with("./") || stem.starts_with("../") {
707            format!("{stem}.md")
708        } else {
709            format!("./{stem}.md")
710        }
711    } else if stripped_prefix.ends_with(".md") {
712        if stripped_prefix.starts_with("./") || stripped_prefix.starts_with("../") {
713            stripped_prefix.to_string()
714        } else {
715            format!("./{stripped_prefix}")
716        }
717    } else {
718        stripped_prefix.to_string()
719    };
720
721    match fragment {
722        Some(f) if !f.is_empty() => format!("{relative_md}#{}", slugify_anchor(f)),
723        _ => relative_md,
724    }
725}
726
727fn strip_version_prefix(path: &str) -> &str {
728    if let Some(rest) = path.strip_prefix("latest/") {
729        return rest;
730    }
731    if let Some(rest) = path.strip_prefix('v') {
732        if let Some((ver, remainder)) = rest.split_once('/') {
733            let is_semver = ver
734                .split('.')
735                .all(|part| !part.is_empty() && part.chars().all(|c| c.is_ascii_digit()));
736            if is_semver {
737                return remainder;
738            }
739        }
740    }
741    path
742}
743
744fn slugify_anchor(anchor: &str) -> String {
745    anchor
746        .trim()
747        .to_ascii_lowercase()
748        .chars()
749        .map(|c| {
750            if c.is_ascii_alphanumeric() || c == '-' || c == '_' {
751                c
752            } else {
753                '-'
754            }
755        })
756        .collect::<String>()
757        .trim_matches('-')
758        .to_string()
759}

761/// Prevents more than two consecutive newlines after stripping block-level HTML wrappers.
762fn collapse_excess_blank_lines(input: &str) -> String {
763    let mut out = String::with_capacity(input.len());
764    let mut consecutive_newlines = 0;
765
766    for ch in input.chars() {
767        if ch == '\n' {
768            consecutive_newlines += 1;
769            if consecutive_newlines <= 2 {
770                out.push(ch);
771            }
772        } else {
773            consecutive_newlines = 0;
774            out.push(ch);
775        }
776    }
777
778    out
779}
