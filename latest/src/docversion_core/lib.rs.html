4#[derive(Debug, Error, PartialEq, Eq)]
5pub enum CoreError {
6    #[error("Invalid SemVer string: {0}")]
7    InvalidVersion(String),
8    #[error("Target path contains forbidden relative traversals: {0}")]
9    UnsafePath(String),
10}
11
12impl From<semver::Error> for CoreError {
13    fn from(err: semver::Error) -> Self {
14        CoreError::InvalidVersion(err.to_string())
15    }
16}
17
18#[derive(Debug, Clone, PartialEq, Eq)]
19pub struct VersionIdentifier {
20    pub declared_version: semver::Version,
21    pub revision_suffix: Option<String>,
22}
23
24impl VersionIdentifier {
25    /// Constructs a new `VersionIdentifier` from a SemVer string and an optional revision suffix.
26    ///
27    /// ```rust
28    /// use docversion_core::VersionIdentifier;
29    ///
30    /// let ver = VersionIdentifier::new("1.4.2", Some("rev.8a4c1f0")).unwrap();
31    /// assert_eq!(ver.display_version(), "v1.4.2 (rev.8a4c1f0)");
32    /// assert_eq!(ver.target_directory(), "v1.4.2");
33    /// ```
34    pub fn new(declared: &str, revision: Option<&str>) -> Result<Self, CoreError> {
35        let declared_version = semver::Version::parse(declared.trim())?;
36        let revision_suffix = revision
37            .map(|s| s.trim())
38            .filter(|s| !s.is_empty())
39            .map(|s| s.to_string());
40
41        Ok(Self {
42            declared_version,
43            revision_suffix,
44        })
45    }
46
47    /// Formats the version for display banners.
48    pub fn display_version(&self) -> String {
49        match &self.revision_suffix {
50            Some(rev) => format!("v{} ({})", self.declared_version, rev),
51            None => format!("v{}", self.declared_version),
52        }
53    }
54
55    /// Resolves the filesystem target directory component.
56    pub fn target_directory(&self) -> String {
57        format!("v{}", self.declared_version)
58    }
59}
60
61impl fmt::Display for VersionIdentifier {
62    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
63        write!(f, "{}", self.display_version())
64    }
65}

67pub fn escape_html_attribute(input: &str) -> String {
68    input
69        .replace('&', "&amp;")
70        .replace('<', "&lt;")
71        .replace('>', "&gt;")
72        .replace('"', "&quot;")
73        .replace('\'', "&#x27;")
74}
75
76/// Generates a validated, escaped HTML redirect targeting `target_url`.
77///
78/// ```rust
79/// use docversion_core::generate_redirect_html;
80///
81/// let markup = generate_redirect_html("latest/docversion/index.html").unwrap();
82/// assert!(markup.contains("latest/docversion/index.html"));
83/// ```
84pub fn generate_redirect_html(target_url: &str) -> Result<String, CoreError> {
85    if target_url.contains("..") || target_url.starts_with('\\') {
86        return Err(CoreError::UnsafePath(target_url.to_string()));
87    }
88
89    let safe_url = escape_html_attribute(target_url);
90    Ok(format!(
91        r#"<!DOCTYPE html>
92<html lang="en">
93  <head>
94    <meta charset="utf-8">
95    <meta http-equiv="refresh" content="0; url={safe_url}">
96    <link rel="canonical" href="{safe_url}">
97    <title>Redirecting...</title>
98  </head>
99  <body>
100    <p>Redirecting to <a href="{safe_url}">{safe_url}</a>...</p>
101  </body>
102</html>
103"#
104    ))
105}
