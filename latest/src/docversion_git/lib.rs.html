4pub struct GitContext {

9impl GitContext {
10    pub fn resolve<P: AsRef<Path>>(repo_path: P) -> Result<Self> {
11        let repo = gix::discover(repo_path.as_ref()).context("Failed to locate Git repository")?;
12
13        let mut head = repo.head().context("Failed to resolve HEAD")?;
14        let head_commit = head
15            .peel_to_commit_in_place()
16            .context("Failed to peel HEAD to commit")?;
17        let tree_hash = head_commit.id().to_hex_with_len(7).to_string();
18
19        let is_dirty = check_is_dirty(&repo);
20
21        Ok(Self {
22            tree_hash,
23            is_dirty,
24        })
25    }
26}

34/// Computes the revision string for the specified paths (e.g. "rev.1a2b3c4" or "rev.1a2b3c4+dirty").
35pub fn compute_doc_revision<P: AsRef<Path>>(repo_path: P, _paths: &[&str]) -> Result<String> {
36    match GitContext::resolve(repo_path) {
37        Ok(ctx) => {
38            if ctx.is_dirty {
39                Ok(format!("rev.{}+dirty", ctx.tree_hash))
40            } else {
41                Ok(format!("rev.{}", ctx.tree_hash))
42            }
43        }
44        Err(_) => Ok("rev.clean".to_string()),
45    }
46}

48fn check_is_dirty(repo: &gix::Repository) -> bool {
49    let work_dir = match repo.work_dir() {
50        Some(w) => w,
51        None => return false,
52    };
53
54    let index = match repo.index_or_empty() {
55        Ok(idx) => idx,
56        Err(_) => return false,
57    };
58
59    for entry in index.entries() {
60        let path = match std::str::from_utf8(entry.path(&index)) {
61            Ok(p) => work_dir.join(p),
62            Err(_) => continue,
63        };
64
65        if let Ok(meta) = std::fs::metadata(&path) {
66            let stat = entry.stat;
67            if stat.size != 0 && stat.size != meta.len() as u32 {
68                return true;
69            }
70        } else {
71            return true;
72        }
73    }
74
75    false
