22/// Context carried through Pass 2 Markdown rendering for exact cross-crate link resolution.
23struct RenderContext<'a> {
24    index: &'a WorkspaceSymbolIndex,
25    current_crate: &'a str,

44/// Generates a GitHub/Shortcut-compatible heading anchor slug matching Markdown heading rules.
45fn heading_slug(raw_heading: &str) -> String {
46    let mut slug = String::with_capacity(raw_heading.len());
47    for ch in raw_heading.chars() {
48        if ch.is_ascii_alphanumeric() {
49            slug.push(ch.to_ascii_lowercase());
50        } else if ch == ' ' || ch == '-' || ch == '_' {
51            slug.push('-');
52        }
53    }
54    while slug.contains("--") {
55        slug = slug.replace("--", "-");
56    }
57    format!("#{}", slug.trim_matches('-'))
58}

60/// Normalizes `quote!` token strings into clean, idiomatic code formatting.
61fn normalize_token_spacing(raw: &str) -> String {
62    let mut s = raw.split_whitespace().collect::<Vec<_>>().join(" ");
63    for (pat, rep) in [
64        (" :: ", "::"),
65        (":: ", "::"),
66        (" ::", "::"),
67        (" < ", "<"),
68        ("< ", "<"),
69        (" <", "<"),
70        (" > ", "> "),
71        (" >", ">"),
72        (" ( ", "("),
73        ("( ", "("),
74        (" (", "("),
75        (" ) ", ") "),
76        (" )", ")"),
77        (" [ ", "["),
78        ("[ ", "["),
79        (" [", "["),
80        (" ]", "]"),
81        (" ,", ","),
82        (" ;", ";"),
83        (" :", ":"),
84        ("& ", "&"),
85        ("->(", "-> ("),
86        (":(", ": ("),
87        ("fn(", "fn ("),
88        ("for(", "for ("),
89        ("if(", "if ("),
90        ("while(", "while ("),
91        ("match(", "match ("),
92        ("in(", "in ("),
93    ] {
94        while s.contains(pat) {
95            s = s.replace(pat, rep);
96        }
97    }
98    s
99}

101/// Extracts doc comments (`///` or `//!`) from attributes.
102fn extract_doc_comments(attrs: &[syn::Attribute]) -> String {
103    let mut docs = Vec::new();
104    for attr in attrs {
105        if attr.path().is_ident("doc") {
106            if let syn::Meta::NameValue(nv) = &attr.meta {
107                if let syn::Expr::Lit(syn::ExprLit {
108                    lit: syn::Lit::Str(s),
109                    ..
110                }) = &nv.value
111                {
112                    docs.push(s.value().trim().to_string());
113                }
114            }
115        }
116    }
117    docs.join("\n")
118}

141fn collect_file_symbols(
142    file_path: &Path,
143    base_dir: &Path,
144    crate_ident: &str,
145    mod_segments: &[String],
146    index: &mut WorkspaceSymbolIndex,
147) -> Result<()> {
148    if !file_path.exists() {
149        return Ok(());
150    }
151
152    let code = fs::read_to_string(file_path)?;
153    let syntax = match syn::parse_file(&code) {
154        Ok(s) => s,
155        Err(_) => return Ok(()),
156    };
157
158    for item in &syntax.items {
159        collect_item_symbol(item, base_dir, crate_ident, mod_segments, index)?;
160    }
161
162    Ok(())
163}

165fn build_qualified_path(crate_ident: &str, mod_segments: &[String], item_name: &str) -> String {
166    if mod_segments.is_empty() {

168    } else {
169        format!("{}::{}::{}", crate_ident, mod_segments.join("::"), item_name)
170    }

181fn collect_item_symbol(
182    item: &Item,
183    base_dir: &Path,
184    crate_ident: &str,
185    mod_segments: &[String],
186    index: &mut WorkspaceSymbolIndex,
187) -> Result<()> {
188    let html_prefix = build_rustdoc_prefix(crate_ident, mod_segments);
189
190    match item {
191        Item::Mod(ItemMod {
192            vis: Visibility::Public(_),
193            ident,
194            content,
195            ..
196        }) => {
197            let name = ident.to_string();
198            let full_path = build_qualified_path(crate_ident, mod_segments, &name);
199            index.insert(SymbolEntry {
200                crate_ident: crate_ident.to_string(),
201                full_path,
202                short_name: name.clone(),
203                kind: SymbolKind::Module,
204                markdown_anchor: heading_slug(&format!("Module {}", name)),
205                rustdoc_path: format!("{}/{}/index.html", html_prefix, name),
206            });
207
208            let mut next_mods = mod_segments.to_vec();
209            next_mods.push(name.clone());
210
211            if let Some((_, inline_items)) = content {
212                for inner in inline_items {
213                    collect_item_symbol(inner, base_dir, crate_ident, &next_mods, index)?;
214                }
215            } else {
216                let mod_file_direct = base_dir.join(format!("{}.rs", name));
217                let mod_file_dir = base_dir.join(&name).join("mod.rs");
218
219                if mod_file_direct.exists() {
220                    collect_file_symbols(
221                        &mod_file_direct,
222                        &base_dir.join(&name),
223                        crate_ident,
224                        &next_mods,
225                        index,
226                    )?;
227                } else if mod_file_dir.exists() {
228                    collect_file_symbols(
229                        &mod_file_dir,
230                        &base_dir.join(&name),
231                        crate_ident,
232                        &next_mods,
233                        index,
234                    )?;
235                }
236            }
237        }
238        Item::Struct(ItemStruct {
239            vis: Visibility::Public(_),
240            ident,
241            ..
242        }) => {
243            let name = ident.to_string();
244            index.insert(SymbolEntry {
245                crate_ident: crate_ident.to_string(),
246                full_path: build_qualified_path(crate_ident, mod_segments, &name),
247                short_name: name.clone(),
248                kind: SymbolKind::Struct,
249                markdown_anchor: heading_slug(&format!("struct {}", name)),
250                rustdoc_path: format!("{}/struct.{}.html", html_prefix, name),
251            });
252        }
253        Item::Enum(ItemEnum {
254            vis: Visibility::Public(_),
255            ident,
256            ..
257        }) => {
258            let name = ident.to_string();
259            index.insert(SymbolEntry {
260                crate_ident: crate_ident.to_string(),
261                full_path: build_qualified_path(crate_ident, mod_segments, &name),
262                short_name: name.clone(),
263                kind: SymbolKind::Enum,
264                markdown_anchor: heading_slug(&format!("enum {}", name)),
265                rustdoc_path: format!("{}/enum.{}.html", html_prefix, name),
266            });
267        }
268        Item::Trait(ItemTrait {
269            vis: Visibility::Public(_),
270            ident,
271            ..
272        }) => {
273            let name = ident.to_string();
274            index.insert(SymbolEntry {
275                crate_ident: crate_ident.to_string(),
276                full_path: build_qualified_path(crate_ident, mod_segments, &name),
277                short_name: name.clone(),
278                kind: SymbolKind::Trait,
279                markdown_anchor: heading_slug(&format!("trait {}", name)),
280                rustdoc_path: format!("{}/trait.{}.html", html_prefix, name),
281            });
282        }
283        Item::Fn(ItemFn {
284            vis: Visibility::Public(_),
285            sig,
286            ..
287        }) => {
288            let name = sig.ident.to_string();
289            index.insert(SymbolEntry {
290                crate_ident: crate_ident.to_string(),
291                full_path: build_qualified_path(crate_ident, mod_segments, &name),
292                short_name: name.clone(),
293                kind: SymbolKind::Function,
294                markdown_anchor: heading_slug(&format!("fn {}", name)),
295                rustdoc_path: format!("{}/fn.{}.html", html_prefix, name),
296            });
297        }
298        Item::Type(ItemType {
299            vis: Visibility::Public(_),
300            ident,
301            ..
302        }) => {
303            let name = ident.to_string();
304            index.insert(SymbolEntry {
305                crate_ident: crate_ident.to_string(),
306                full_path: build_qualified_path(crate_ident, mod_segments, &name),
307                short_name: name.clone(),
308                kind: SymbolKind::TypeAlias,
309                markdown_anchor: heading_slug(&format!("type {}", name)),
310                rustdoc_path: format!("{}/type.{}.html", html_prefix, name),
311            });
312        }
313        Item::Const(ItemConst {
314            vis: Visibility::Public(_),
315            ident,
316            ..
317        }) => {
318            let name = ident.to_string();
319            index.insert(SymbolEntry {
320                crate_ident: crate_ident.to_string(),
321                full_path: build_qualified_path(crate_ident, mod_segments, &name),
322                short_name: name.clone(),
323                kind: SymbolKind::Constant,
324                markdown_anchor: heading_slug(&format!("const {}", name)),
325                rustdoc_path: format!("{}/constant.{}.html", html_prefix, name),
326            });
327        }
328        Item::Impl(ItemImpl {
329            items,
330            self_ty,
331            trait_,
332            ..
333        }) => {
334            let type_name = normalize_token_spacing(&quote::quote!(#self_ty).to_string());
335            let is_trait_impl = trait_.is_some();
336            let impl_header = if let Some((_, tr, _)) = trait_ {
337                let tr_name = normalize_token_spacing(&quote::quote!(#tr).to_string());
338                format!("impl {} for {}", tr_name, type_name)
339            } else {
340                format!("impl {}", type_name)
341            };
342
343            for impl_item in items {
344                if let ImplItem::Fn(m) = impl_item {
345                    if is_trait_impl || matches!(m.vis, Visibility::Public(_)) {
346                        let method_name = m.sig.ident.to_string();
347                        let scoped_short = format!("{}::{}", type_name, method_name);
348                        let full_path =
349                            build_qualified_path(crate_ident, mod_segments, &scoped_short);
350                        let heading_text = format!("fn {} ({})", method_name, impl_header);
351
352                        index.insert(SymbolEntry {
353                            crate_ident: crate_ident.to_string(),
354                            full_path,
355                            short_name: scoped_short,
356                            kind: SymbolKind::Method,
357                            markdown_anchor: heading_slug(&heading_text),
358                            rustdoc_path: format!(
359                                "{}/struct.{}.html#method.{}",
360                                html_prefix, type_name, method_name
361                            ),
362                        });
363                    }
364                }
365            }
366        }
367        _ => {}
368    }
369
370    Ok(())
371}

373/// Scans a single crate root (`src/lib.rs` or `src/main.rs`) and appends its public API markdown.
374fn scan_crate_root(
375    crate_name: &str,
376    crate_root: &Path,
377    out: &mut String,
378    crate_heading_level: usize,
379    ctx: &RenderContext<'_>,
380) -> Result<()> {
381    let hashes = "#".repeat(crate_heading_level.min(6));
382    out.push_str(&format!("---\n\n{} Crate `{}`\n\n", hashes, crate_name));
383
384    let src_dir = crate_root.join("src");
385    let lib_rs = src_dir.join("lib.rs");
386    let main_rs = src_dir.join("main.rs");
387
388    let entry_file = if lib_rs.exists() {
389        lib_rs
390    } else if main_rs.exists() {
391        main_rs
392    } else {
393        return Ok(());
394    };
395
396    scan_file(&entry_file, &src_dir, out, crate_heading_level + 1, ctx)
397}

399/// Two-pass workspace scanner:
400/// 1. Indexes all public symbols across workspace crates into `WorkspaceSymbolIndex`.
401/// 2. Renders public interfaces, doctests, multi-language files, and `docs/*.md` with
402///    resolved cross-crate intra-doc links.
403pub fn generate_markdown_summary(workspace_title: &str, crate_root: &Path) -> Result<String> {
404    let mut symbol_index = WorkspaceSymbolIndex::new();
405    let mut workspace_crates: Vec<(String, String, PathBuf)> = Vec::new();
406
407    let manifest_path = crate_root.join("Cargo.toml");
408    if manifest_path.exists() {
409        if let Ok(metadata) = MetadataCommand::new()
410            .manifest_path(&manifest_path)
411            .no_deps()
412            .exec()
413        {
414            let mut packages = metadata.workspace_packages();
415            packages.sort_by(|a, b| a.name.cmp(&b.name));
416
417            for pkg in packages {
418                if let Some(pkg_dir) = pkg.manifest_path.parent() {
419                    let crate_ident = pkg.name.replace('-', "_");
420                    let pkg_path = pkg_dir.as_std_path().to_path_buf();
421                    collect_crate_symbols(&crate_ident, &pkg_path, &mut symbol_index)?;
422                    workspace_crates.push((pkg.name.clone(), crate_ident, pkg_path));
423                }
424            }
425        }
426    }
427
428    if workspace_crates.is_empty() {
429        let fallback_ident = workspace_title.replace('-', "_");
430        collect_crate_symbols(&fallback_ident, crate_root, &mut symbol_index)?;
431    }
432
433    let mut out = String::new();
434    out.push_str(&format!("# {}\n\n", workspace_title));
435    out.push_str("> Autonomous Agent API Context & Single-File Documentation Index\n\n");
436
437    if !workspace_crates.is_empty() {
438        for (crate_name, crate_ident, pkg_dir) in &workspace_crates {
439            let ctx = RenderContext {
440                index: &symbol_index,
441                current_crate: crate_ident,
442                current_type: None,
443            };
444            scan_crate_root(crate_name, pkg_dir, &mut out, 2, &ctx)?;
445        }
446    } else {
447        let fallback_ident = workspace_title.replace('-', "_");
448        let ctx = RenderContext {
449            index: &symbol_index,
450            current_crate: &fallback_ident,
451            current_type: None,
452        };
453        let src_dir = crate_root.join("src");
454        let lib_rs = src_dir.join("lib.rs");
455        let main_rs = src_dir.join("main.rs");
456
457        if lib_rs.exists() {
458            scan_file(&lib_rs, &src_dir, &mut out, 2, &ctx)?;
459        } else if main_rs.exists() {
460            scan_file(&main_rs, &src_dir, &mut out, 2, &ctx)?;
461        }
462    }
463
464    // Universal multi-language source scanner (.ts, .tsx, .js, .py, .go, .sql)
465    scan_universal_sources(crate_root, crate_root, &mut out)?;
466
467    // Documentation directory Markdown ingestion (docs/**/*.md)
468    let docs_dir = crate_root.join("docs");
469    if docs_dir.is_dir() {
470        let default_crate = workspace_crates
471            .first()
472            .map(|(_, ident, _)| ident.as_str())
473            .unwrap_or("docversion_core");
474        let ctx = RenderContext {
475            index: &symbol_index,
476            current_crate: default_crate,
477            current_type: None,
478        };
479        scan_markdown_docs(&docs_dir, crate_root, &mut out, &ctx)?;
480    }
481
482    Ok(out)
483}

485/// Walks `root` for non-Rust source files and extracts public interfaces + docstrings.
486fn scan_universal_sources(dir: &Path, workspace_root: &Path, out: &mut String) -> Result<()> {
487    if !dir.is_dir() {
488        return Ok(());
489    }
490    let mut entries: Vec<_> = fs::read_dir(dir)?.filter_map(|e| e.ok()).collect();
491    entries.sort_by_key(|e| e.file_name());
492
493    for entry in entries {
494        let path = entry.path();
495        let name = entry.file_name();
496        let name_str = name.to_string_lossy();
497
498        if path.is_dir() {
499            if matches!(
500                name_str.as_ref(),
501                "target" | "node_modules" | ".git" | ".venv" | "venv" | "__pycache__" | "dist"
502            ) {
503                continue;
504            }
505            scan_universal_sources(&path, workspace_root, out)?;
506        } else if let Some(ext) = path.extension().and_then(|e| e.to_str()) {
507            let lang = match ext {
508                "ts" | "tsx" => Some("typescript"),
509                "js" | "jsx" => Some("javascript"),
510                "py" => Some("python"),
511                "go" => Some("go"),
512                "sql" => Some("sql"),
513                _ => None,
514            };
515            if let Some(lang_tag) = lang {
516                let rel = path.strip_prefix(workspace_root).unwrap_or(&path);
517                let content = fs::read_to_string(&path).unwrap_or_default();
518                let decls = extract_universal_declarations(&content, lang_tag);
519                if !decls.is_empty() {
520                    out.push_str(&format!("---\n\n## File `{}`\n\n", rel.display()));
521                    for (doc, sig) in decls {
522                        if !doc.is_empty() {
523                            out.push_str(&format!("{}\n\n", doc));
524                        }
525                        out.push_str(&format!("~~~{}\n{}\n~~~\n\n", lang_tag, sig));
526                    }
527                }
528            }
529        }
530    }
531    Ok(())
532}

534/// Extracts top-level exported/public declarations and preceding docstrings across TS/JS, Python, Go, and SQL.
535fn extract_universal_declarations(content: &str, lang: &str) -> Vec<(String, String)> {
536    let mut results = Vec::new();
537    let lines: Vec<&str> = content.lines().collect();
538
539    for (idx, line) in lines.iter().enumerate() {
540        let trimmed = line.trim();
541        let is_decl = match lang {
542            "typescript" | "javascript" => {
543                trimmed.starts_with("export function ")
544                    || trimmed.starts_with("export async function ")
545                    || trimmed.starts_with("export interface ")
546                    || trimmed.starts_with("export type ")
547                    || trimmed.starts_with("export class ")
548                    || trimmed.starts_with("export enum ")
549            }
550            "python" => {
551                (line.starts_with("def ")
552                    || line.starts_with("async def ")
553                    || line.starts_with("class "))
554                    && !trimmed.starts_with("def _")
555                    && !trimmed.starts_with("async def _")
556            }
557            "go" => {
558                (trimmed.starts_with("func ") || trimmed.starts_with("type "))
559                    && is_go_exported_decl(trimmed)
560            }
561            "sql" => {
562                let upper = trimmed.to_ascii_uppercase();
563                upper.starts_with("CREATE TABLE ")
564                    || upper.starts_with("CREATE VIEW ")
565                    || upper.starts_with("CREATE INDEX ")
566            }
567            _ => false,
568        };
569
570        if is_decl {
571            let mut doc_lines = Vec::new();
572            let mut back = idx;
573            while back > 0 {
574                back -= 1;
575                let prev = lines[back].trim();
576                if let Some(comment) = prev
577                    .strip_prefix("///")
578                    .or_else(|| prev.strip_prefix("//"))
579                    .or_else(|| prev.strip_prefix('#'))
580                    .or_else(|| prev.strip_prefix("--"))
581                    .or_else(|| prev.strip_prefix('*'))
582                {
583                    let cleaned = comment.trim_end_matches("*/").trim();
584                    if !cleaned.is_empty() {
585                        doc_lines.push(cleaned.to_string());
586                    }
587                } else if prev == "/**" || prev == "/*" {
588                    continue;
589                } else {
590                    break;
591                }
592            }
593            doc_lines.reverse();
594
595            let sig = if lang == "sql" {
596                let mut stmt = Vec::new();
597                for future_line in &lines[idx..] {
598                    stmt.push(future_line.trim_end());
599                    if future_line.contains(';') {
600                        break;
601                    }
602                }
603                stmt.join("\n")
604            } else {
605                trimmed.trim_end_matches('{').trim().to_string()
606            };
607
608            results.push((doc_lines.join(" "), sig));
609        }
610    }
611
612    results
613}

615fn is_go_exported_decl(trimmed: &str) -> bool {
616    let after_kw = trimmed
617        .strip_prefix("func ")
618        .or_else(|| trimmed.strip_prefix("type "))
619        .unwrap_or("");
620    let ident_part = if let Some(after_recv) = after_kw.strip_prefix('(') {
621        after_recv
622            .split_once(')')
623            .map(|(_, rest)| rest.trim_start())
624            .unwrap_or(after_kw)
625    } else {
626        after_kw
627    };
628    ident_part.chars().next().is_some_and(|c| c.is_ascii_uppercase())
629}

631/// Appends authored Markdown files from `docs/` into the single-file bundle with resolved symbol links.
632fn scan_markdown_docs(
633    dir: &Path,
634    workspace_root: &Path,
635    out: &mut String,
636    ctx: &RenderContext<'_>,
637) -> Result<()> {
638    let mut entries: Vec<_> = fs::read_dir(dir)?.filter_map(|e| e.ok()).collect();
639    entries.sort_by_key(|e| e.file_name());
640
641    for entry in entries {
642        let path = entry.path();
643        if path.is_dir() {
644            scan_markdown_docs(&path, workspace_root, out, ctx)?;
645        } else if path.extension().is_some_and(|ext| ext == "md") {
646            let rel = path.strip_prefix(workspace_root).unwrap_or(&path);
647            let content = fs::read_to_string(&path).unwrap_or_default();
648            if !content.trim().is_empty() {
649                let resolved = ctx.resolve_text(content.trim());
650                out.push_str(&format!("---\n\n## Guide `{}`\n\n", rel.display()));
651                out.push_str(&resolved);
652                out.push_str("\n\n");
653            }
654        }
655    }
656    Ok(())
657}

659fn scan_file(
660    file_path: &Path,
661    base_dir: &Path,
662    out: &mut String,
663    heading_level: usize,
664    ctx: &RenderContext<'_>,
665) -> Result<()> {
666    if !file_path.exists() {
667        return Ok(());
668    }
669
670    let code = fs::read_to_string(file_path)?;
671    let syntax = match syn::parse_file(&code) {
672        Ok(s) => s,
673        Err(_) => return Ok(()),
674    };
675
676    let file_docs = extract_doc_comments(&syntax.attrs);
677    if !file_docs.is_empty() {
678        let contract = parse_intent_contract(&file_docs);
679        format_intent_markdown(out, &contract, heading_level, ctx);
680    }
681
682    for item in syntax.items {
683        process_item(&item, base_dir, out, heading_level, ctx)?;
684    }
685
686    Ok(())
687}

689#[derive(Debug, Clone)]
690pub enum InvariantKind {
691    Assert(String),
692    ErrorRejection,
693}
694
695/// Extracted runtime invariant check, assertion, or error rejection boundary.
696#[derive(Debug, Clone)]
697pub struct ExtractedAssertion {
698    pub kind: InvariantKind,
699    pub condition: String,
700}
701
702pub struct AssertionVisitor {
703    pub assertions: Vec<ExtractedAssertion>,
704}
705
706impl Default for AssertionVisitor {
707    fn default() -> Self {
708        Self::new()
709    }
710}
711
712impl AssertionVisitor {
713    pub fn new() -> Self {
714        Self {
715            assertions: Vec::new(),
716        }
717    }
718}
719
720impl<'ast> Visit<'ast> for AssertionVisitor {
721    fn visit_expr_macro(&mut self, node: &'ast ExprMacro) {
722        if let Some(ident) = node.mac.path.get_ident() {
723            let name = ident.to_string();
724            if matches!(
725                name.as_str(),
726                "assert" | "assert_eq" | "assert_ne" | "ensure" | "debug_assert"
727            ) {
728                let tokens = node.mac.tokens.to_string();
729                let clean = normalize_token_spacing(&tokens);
730                if !self
731                    .assertions
732                    .iter()
733                    .any(|a| matches!(&a.kind, InvariantKind::Assert(k) if k == &name) && a.condition == clean)
734                {
735                    self.assertions.push(ExtractedAssertion {
736                        kind: InvariantKind::Assert(name),
737                        condition: clean,
738                    });
739                }
740            }
741        }
742        visit::visit_expr_macro(self, node);
743    }
744
745    fn visit_expr_call(&mut self, node: &'ast ExprCall) {
746        if let Some(err_desc) = match_err_call(node) {
747            if !self
748                .assertions
749                .iter()
750                .any(|a| matches!(a.kind, InvariantKind::ErrorRejection) && a.condition == err_desc)
751            {
752                self.assertions.push(ExtractedAssertion {
753                    kind: InvariantKind::ErrorRejection,
754                    condition: err_desc,
755                });
756            }
757        }
758        visit::visit_expr_call(self, node);
759    }
760}
761
762fn match_err_call(call: &ExprCall) -> Option<String> {
763    if let syn::Expr::Path(p) = &*call.func {
764        if let Some(seg) = p.path.segments.last() {
765            if seg.ident == "Err" {
766                let tokens = quote::quote!(#call).to_string();
767                return Some(normalize_token_spacing(&tokens));
768            }
769        }
770    }
771    None
772}

774/// Scans an AST block for runtime assertions and invariant guards.
775fn extract_block_assertions(block: &syn::Block) -> Vec<ExtractedAssertion> {
776    let mut visitor = AssertionVisitor::new();
777    visitor.visit_block(block);
778    visitor.assertions
779}

781pub fn format_assertions_markdown(
782    out: &mut String,
783    assertions: &[ExtractedAssertion],
784    heading_level: usize,
785) {
786    if assertions.is_empty() {
787        return;
788    }
789
790    let subhashes = "#".repeat((heading_level + 1).min(6));
791    out.push_str(&format!(
792        "{} **Programmatic Invariants & Rejection Boundaries**\n\n",
793        subhashes
794    ));
795
796    for a in assertions {
797        match &a.kind {
798            InvariantKind::Assert(name) => {
799                out.push_str(&format!("* 🔒 `{}`: `{}`\n", name, a.condition));
800            }
801            InvariantKind::ErrorRejection => {
802                out.push_str(&format!("* 🚫 Reject: `{}`\n", a.condition));
803            }
804        }
805    }
806    out.push('\n');
807}

809fn process_item(
810    item: &Item,
811    base_dir: &Path,
812    out: &mut String,
813    level: usize,
814    ctx: &RenderContext<'_>,
815) -> Result<()> {
816    let hashes = "#".repeat(level.min(6));
817
818    match item {
819        Item::Mod(ItemMod {
820            vis: Visibility::Public(_),
821            ident,
822            content,
823            attrs,
824            ..
825        }) => {
826            out.push_str(&format!("{} Module `{}`\n\n", hashes, ident));
827            let docs = extract_doc_comments(attrs);
828            if !docs.is_empty() {
829                let contract = parse_intent_contract(&docs);
830                format_intent_markdown(out, &contract, level, ctx);
831            }
832
833            if let Some((_, inline_items)) = content {
834                for inner in inline_items {
835                    process_item(inner, base_dir, out, level + 1, ctx)?;
836                }
837            } else {
838                let mod_file_direct = base_dir.join(format!("{}.rs", ident));
839                let mod_file_dir = base_dir.join(ident.to_string()).join("mod.rs");
840
841                if mod_file_direct.exists() {
842                    scan_file(
843                        &mod_file_direct,
844                        &base_dir.join(ident.to_string()),
845                        out,
846                        level + 1,
847                        ctx,
848                    )?;
849                } else if mod_file_dir.exists() {
850                    scan_file(
851                        &mod_file_dir,
852                        &base_dir.join(ident.to_string()),
853                        out,
854                        level + 1,
855                        ctx,
856                    )?;
857                }
858            }
859        }
860        Item::Const(ItemConst {
861            vis: Visibility::Public(_),
862            ident,
863            ty,
864            expr,
865            attrs,
866            ..
867        }) => {
868            out.push_str(&format!("{} `const {}`\n\n", hashes, ident));
869            let docs = extract_doc_comments(attrs);
870            if !docs.is_empty() {
871                let contract = parse_intent_contract(&docs);
872                format_intent_markdown(out, &contract, level, ctx);
873            }
874            let ty_str = normalize_token_spacing(&quote::quote!(#ty).to_string());
875            let expr_str = normalize_token_spacing(&quote::quote!(#expr).to_string());
876            out.push_str(&format!(
877                "~~~rust\npub const {}: {} = {};\n~~~\n\n",
878                ident, ty_str, expr_str
879            ));
880        }
881        Item::Type(ItemType {
882            vis: Visibility::Public(_),
883            ident,
884            generics,
885            ty,
886            attrs,
887            ..
888        }) => {
889            out.push_str(&format!("{} `type {}`\n\n", hashes, ident));
890            let docs = extract_doc_comments(attrs);
891            if !docs.is_empty() {
892                let contract = parse_intent_contract(&docs);
893                format_intent_markdown(out, &contract, level, ctx);
894            }
895            let gen_str = normalize_token_spacing(&quote::quote!(#generics).to_string());
896            let ty_str = normalize_token_spacing(&quote::quote!(#ty).to_string());
897            out.push_str(&format!(
898                "~~~rust\npub type {}{} = {};\n~~~\n\n",
899                ident, gen_str, ty_str
900            ));
901        }
902        Item::Struct(ItemStruct {
903            vis: Visibility::Public(_),
904            ident,
905            fields,
906            attrs,
907            ..
908        }) => {
909            let struct_name = ident.to_string();
910            let scoped_ctx = ctx.with_type(&struct_name);
911            out.push_str(&format!("{} `struct {}`\n\n", hashes, ident));
912            let docs = extract_doc_comments(attrs);
913            if !docs.is_empty() {
914                let contract = parse_intent_contract(&docs);
915                format_intent_markdown(out, &contract, level, &scoped_ctx);
916            }
917
918            match fields {
919                Fields::Named(named) => {
920                    out.push_str("**Fields:**\n\n");
921                    for f in &named.named {
922                        if let Visibility::Public(_) = f.vis {
923                            let f_name =
924                                f.ident.as_ref().map(|i| i.to_string()).unwrap_or_default();
925                            let f_ty_raw = &f.ty;
926                            let f_ty = normalize_token_spacing(&quote::quote!(#f_ty_raw).to_string());
927                            let f_doc = extract_doc_comments(&f.attrs);
928                            if f_doc.is_empty() {
929                                out.push_str(&format!("* `{}: {}`\n", f_name, f_ty));
930                            } else {
931                                let resolved_doc = scoped_ctx.resolve_text(&f_doc);
932                                out.push_str(&format!(
933                                    "* `{}: {}` - {}\n",
934                                    f_name, f_ty, resolved_doc
935                                ));
936                            }
937                        }
938                    }
939                    out.push('\n');
940                }
941                Fields::Unnamed(unnamed) => {
942                    let tuple_types: Vec<_> = unnamed
943                        .unnamed
944                        .iter()
945                        .map(|f| {
946                            let ty = &f.ty;
947                            normalize_token_spacing(&quote::quote!(#ty).to_string())
948                        })
949                        .collect();
950                    out.push_str(&format!("Tuple Struct: `({})`\n\n", tuple_types.join(", ")));
951                }
952                Fields::Unit => {}
953            }
954        }
955        Item::Enum(ItemEnum {
956            vis: Visibility::Public(_),
957            ident,
958            variants,
959            attrs,
960            ..
961        }) => {
962            let enum_name = ident.to_string();
963            let scoped_ctx = ctx.with_type(&enum_name);
964            out.push_str(&format!("{} `enum {}`\n\n", hashes, ident));
965            let docs = extract_doc_comments(attrs);
966            if !docs.is_empty() {
967                let contract = parse_intent_contract(&docs);
968                format_intent_markdown(out, &contract, level, &scoped_ctx);
969            }
970
971            out.push_str("**Variants:**\n\n");
972            for v in variants {
973                let v_doc = extract_doc_comments(&v.attrs);
974                let v_name = &v.ident;
975                if v_doc.is_empty() {
976                    out.push_str(&format!("* `{}`\n", v_name));
977                } else {
978                    let resolved_doc = scoped_ctx.resolve_text(&v_doc);
979                    out.push_str(&format!("* `{}` - {}\n", v_name, resolved_doc));
980                }
981            }
982            out.push('\n');
983        }
984        Item::Trait(ItemTrait {
985            vis: Visibility::Public(_),
986            ident,
987            items,
988            attrs,
989            ..
990        }) => {
991            let trait_name = ident.to_string();
992            let scoped_ctx = ctx.with_type(&trait_name);
993            out.push_str(&format!("{} `trait {}`\n\n", hashes, ident));
994            let docs = extract_doc_comments(attrs);
995            if !docs.is_empty() {
996                let contract = parse_intent_contract(&docs);
997                format_intent_markdown(out, &contract, level, &scoped_ctx);
998            }
999
1000            out.push_str("**Methods:**\n\n");
1001            for ti in items {
1002                if let syn::TraitItem::Fn(m) = ti {
1003                    let m_sig = &m.sig;
1004                    let sig = normalize_token_spacing(&quote::quote!(#m_sig).to_string());
1005                    let m_docs = extract_doc_comments(&m.attrs);
1006                    if m_docs.is_empty() {
1007                        out.push_str(&format!("* `{}`\n", sig));
1008                    } else {
1009                        let resolved_doc = scoped_ctx.resolve_text(&m_docs);
1010                        out.push_str(&format!("* `{}` - {}\n", sig, resolved_doc));
1011                    }
1012                }
1013            }
1014            out.push('\n');
1015        }
1016        Item::Fn(ItemFn {
1017            vis: Visibility::Public(_),
1018            sig,
1019            attrs,
1020            block,
1021            ..
1022        }) => {
1023            out.push_str(&format!("{} `fn {}`\n\n", hashes, sig.ident));
1024            let docs = extract_doc_comments(attrs);
1025            if !docs.is_empty() {
1026                let contract = parse_intent_contract(&docs);
1027                format_intent_markdown(out, &contract, level, ctx);
1028            }
1029            let clean_sig = normalize_token_spacing(&quote::quote!(#sig).to_string());
1030            out.push_str(&format!("~~~rust\npub {}\n~~~\n\n", clean_sig));
1031
1032            let assertions = extract_block_assertions(block);
1033            format_assertions_markdown(out, &assertions, level);
1034        }
1035        Item::Impl(ItemImpl {
1036            items,
1037            self_ty,
1038            trait_,
1039            ..
1040        }) => {
1041            let type_name = normalize_token_spacing(&quote::quote!(#self_ty).to_string());
1042            let scoped_ctx = ctx.with_type(&type_name);
1043            let is_trait_impl = trait_.is_some();
1044            let impl_header = if let Some((_, tr, _)) = trait_ {
1045                let tr_name = normalize_token_spacing(&quote::quote!(#tr).to_string());
1046                format!("impl {} for {}", tr_name, type_name)
1047            } else {
1048                format!("impl {}", type_name)
1049            };
1050
1051            for impl_item in items {
1052                if let ImplItem::Fn(m) = impl_item {
1053                    if is_trait_impl || matches!(m.vis, Visibility::Public(_)) {
1054                        let m_sig = &m.sig;
1055                        let sig_str = normalize_token_spacing(&quote::quote!(#m_sig).to_string());
1056                        out.push_str(&format!(
1057                            "{} `fn {}` ({})\n\n",
1058                            hashes, m.sig.ident, impl_header
1059                        ));
1060
1061                        let docs = extract_doc_comments(&m.attrs);
1062                        if !docs.is_empty() {
1063                            let contract = parse_intent_contract(&docs);
1064                            format_intent_markdown(out, &contract, level, &scoped_ctx);
1065                        }
1066
1067                        let vis_prefix = if is_trait_impl { "" } else { "pub " };
1068                        out.push_str(&format!("~~~rust\n{}{}\n~~~\n\n", vis_prefix, sig_str));
1069
1070                        let assertions = extract_block_assertions(&m.block);
1071                        format_assertions_markdown(out, &assertions, level);
1072                    }
1073                }
1074            }
1075        }
1076        _ => {}
1077    }
1078
1079    Ok(())
1080}

1880fn replace_delimited(content: &str, start_tag: &str, end_delim: &str, replacement: &str) -> String {
1881    let mut result = String::new();
1882    let mut remainder = content;
1883
1884    while let Some(idx) = remainder.find(start_tag) {
1885        result.push_str(&remainder[..idx]);
1886        result.push_str(start_tag);
1887        let after = &remainder[idx + start_tag.len()..];
1888        if let Some(end_idx) = after.find(end_delim) {
1889            result.push_str(replacement);
1890            result.push_str(end_delim);
1891            remainder = &after[end_idx + end_delim.len()..];
1892        } else {
1893            remainder = after;
1894            break;
1895        }
1896    }
1897    result.push_str(remainder);
1898    result
1899}

1991            copy_dir_recursive(&entry_path, &target_path)?;
1992        } else {
1993            fs::copy(&entry_path, &target_path)?;
1994        }
1995    }
1996    Ok(())
1997}

2008#[derive(Default, Debug)]
2009struct IntentContract {
2010    summary: String,
2011    doctests: Vec<String>,
2012    intent: Option<String>,
2013    guarantees: Vec<String>,
2014    invariants: Vec<String>,
2015    non_goals: Vec<String>,
2016    notes: Vec<String>,
2017}

2019fn parse_intent_contract(doc_text: &str) -> IntentContract {
2020    let mut contract = IntentContract::default();
2021    let mut target: Option<&mut Vec<String>> = None;
2022    let mut in_fence = false;
2023    let mut current_fence = String::new();
2024
2025    for line in doc_text.lines() {
2026        let trimmed_raw = line.trim();
2027        if trimmed_raw.starts_with("```") || trimmed_raw.starts_with("~~~") {
2028            if !in_fence {
2029                in_fence = true;
2030                current_fence.clear();
2031                let lang = trimmed_raw.trim_start_matches(['`', '~']).trim();
2032                let lang_tag = if lang.is_empty() { "rust" } else { lang };
2033                current_fence.push_str(&format!("~~~{}\n", lang_tag));
2034            } else {
2035                in_fence = false;
2036                current_fence.push_str("~~~");
2037                contract.doctests.push(current_fence.clone());
2038                current_fence.clear();
2039            }
2040            continue;
2041        }
2042
2043        if in_fence {
2044            current_fence.push_str(line);
2045            current_fence.push('\n');
2046            continue;
2047        }
2048
2049        let clean = trimmed_raw
2050            .trim_start_matches(['>', '#', '*', '-', ' '])
2051            .trim();
2052        let lower = clean.to_lowercase();
2053
2054        if lower.starts_with("intent") {
2055            let desc = clean["intent".len()..]
2056                .trim_start_matches([':', '*', ' '])
2057                .trim();
2058            if !desc.is_empty() {
2059                contract.intent = Some(desc.to_string());
2060            }
2061            target = None;
2062        } else if lower.starts_with("guarantee") {
2063            target = Some(&mut contract.guarantees);
2064        } else if lower.starts_with("invariant") {
2065            target = Some(&mut contract.invariants);
2066        } else if lower.starts_with("non-goal") || lower.starts_with("non goal") {
2067            target = Some(&mut contract.non_goals);
2068        } else if lower.starts_with("note") {
2069            target = Some(&mut contract.notes);
2070        } else if let Some(ref mut list) = target {
2071            if !clean.is_empty() {
2072                list.push(clean.to_string());
2073            }
2074        } else if contract.intent.is_none() && !clean.is_empty() {
2075            if contract.summary.is_empty() {
2076                contract.summary = clean.to_string();
2077            } else {
2078                contract.summary.push_str(&format!(" {}", clean));
2079            }
2080        }
2081    }
2082
2083    contract
2084}

2086fn format_intent_markdown(
2087    out: &mut String,
2088    contract: &IntentContract,
2089    heading_level: usize,
2090    ctx: &RenderContext<'_>,
2091) {
2092    let subhashes = "#".repeat((heading_level + 1).min(6));
2093
2094    if let Some(intent) = &contract.intent {
2095        let resolved_intent = ctx.resolve_text(intent);
2096        out.push_str(&format!(
2097            "{} **Intent Directive**\n\n> {}\n\n",
2098            subhashes,
2099            resolved_intent.replace('\n', "\n> ")
2100        ));
2101    } else if !contract.summary.is_empty() {
2102        let resolved_summary = ctx.resolve_text(&contract.summary);
2103        out.push_str(&format!("{}\n\n", resolved_summary));
2104    }
2105
2106    for dt in &contract.doctests {
2107        out.push_str(&format!("{}\n\n", dt));
2108    }
2109
2110    if !contract.guarantees.is_empty() {
2111        out.push_str(&format!("{} **Guarantees**\n\n", subhashes));
2112        for g in &contract.guarantees {
2113            out.push_str(&format!("* [x] **Guarantee:** {}\n", ctx.resolve_text(g)));
2114        }
2115        out.push('\n');
2116    }
2117
2118    if !contract.invariants.is_empty() {
2119        out.push_str(&format!("{} **System Invariants**\n\n", subhashes));
2120        for inv in &contract.invariants {
2121            out.push_str(&format!("* 🛡️ `{}`\n", inv));
2122        }
2123        out.push('\n');
2124    }
2125
2126    if !contract.non_goals.is_empty() {
2127        out.push_str(&format!("{} **Explicit Non-Goals**\n\n", subhashes));
2128        for ng in &contract.non_goals {
2129            out.push_str(&format!("* ⛔ {}\n", ctx.resolve_text(ng)));
2130        }
2131        out.push('\n');
2132    }
2133
2134    if !contract.notes.is_empty() {
2135        out.push_str(&format!("{} **Architectural Notes**\n\n", subhashes));
2136        for note in &contract.notes {
2137            out.push_str(&format!("* 📝 {}\n", ctx.resolve_text(note)));
2138        }
2139        out.push('\n');
2140    }
2141}
