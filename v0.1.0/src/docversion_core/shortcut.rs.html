6/// Sanitizes generated Markdown for zero-friction import into Shortcut Docs.
7///
8/// - Preserves fenced code blocks (` ``` ` and `~~~`) and inline backtick spans verbatim.
9/// - Converts `<br>` / `<br/>` into Markdown hard line breaks (`  \n`).
10/// - Converts `<details>` / `<summary>` blocks into clean Markdown bold headers.
11/// - Strips all other raw HTML tags (`<div>`, `<span>`, `<a name="...">`, etc.) while keeping inner text.
12/// - Rewrites `.html` and root-prefixed (`/latest/...`, `/vX.Y.Z/...`) links to relative `.md` paths.
13pub fn sanitize_for_shortcut_docs(markdown: &str) -> String {
14    let mut output = String::with_capacity(markdown.len());
15    let mut in_fence = false;
16    let mut fence_marker: Option<&str> = None;
17
18    for line in markdown.split_inclusive('\n') {
19        let trimmed = line.trim_start();
20
21        if let Some(marker) = detect_fence_prefix(trimmed) {
22            if !in_fence {
23                in_fence = true;
24                fence_marker = Some(marker);
25            } else if fence_marker == Some(marker) {
26                in_fence = false;
27                fence_marker = None;
28            }
29            output.push_str(line);
30            continue;
31        }
32
33        if in_fence {
34            output.push_str(line);
35            continue;
36        }
37
38        let processed = process_prose_line(line);
39        output.push_str(&processed);
40    }
41
42    collapse_excess_blank_lines(&output)
43}

45fn detect_fence_prefix(trimmed_line: &str) -> Option<&'static str> {
46    if trimmed_line.starts_with("```") {
47        Some("```")
48    } else if trimmed_line.starts_with("~~~") {
49        Some("~~~")
50    } else {
51        None
52    }
53}

55/// Processes a single non-fenced Markdown line, respecting inline code spans (`...`).
56fn process_prose_line(line: &str) -> String {
57    let mut result = String::with_capacity(line.len());
58    let mut chars = line.char_indices().peekable();
59    let mut in_inline_code = false;
60    let mut inline_ticks = 0usize;
61    let mut prose_buf = String::new();
62
63    while let Some((_idx, ch)) = chars.next() {
64        if ch == '`' {
65            let mut tick_count = 1;
66            while let Some(&(_, '`')) = chars.peek() {
67                chars.next();
68                tick_count += 1;
69            }
70
71            if !in_inline_code {
72                if !prose_buf.is_empty() {
73                    result.push_str(&transform_prose_segment(&prose_buf));
74                    prose_buf.clear();
75                }
76                in_inline_code = true;
77                inline_ticks = tick_count;
78            } else if tick_count == inline_ticks {
79                in_inline_code = false;
80                inline_ticks = 0;
81            }
82
83            for _ in 0..tick_count {
84                result.push('`');
85            }
86        } else if in_inline_code {
87            result.push(ch);
88        } else {
89            prose_buf.push(ch);
90        }
91    }
92
93    if !prose_buf.is_empty() {
94        result.push_str(&transform_prose_segment(&prose_buf));
95    }
96
97    result
98}

100/// Strips HTML tags and rewrites links inside a pure-prose segment (outside code spans).
101fn transform_prose_segment(segment: &str) -> String {
102    let no_html = strip_html_tags(segment);
103    rewrite_markdown_links(&no_html)
104}

106/// Strips raw HTML tags while converting structural tags (`<br>`, `<summary>`) to Markdown.
107fn strip_html_tags(input: &str) -> String {
108    let mut out = String::with_capacity(input.len());
109    let bytes = input.as_bytes();
110    let mut i = 0;
111
112    while i < bytes.len() {
113        if bytes[i] == b'<' {
114            if let Some(&next) = bytes.get(i + 1) {
115                let is_tag_start =
116                    next == b'/' || next == b'!' || (next as char).is_ascii_alphabetic();
117
118                if is_tag_start {
119                    if let Some(close_offset) = input[i..].find('>') {
120                        let raw_tag = &input[i..=i + close_offset];
121                        let replacement = map_structural_tag(raw_tag);
122                        out.push_str(replacement);
123                        i += close_offset + 1;
124                        continue;
125                    }
126                }
127            }
128        }
129
130        let ch = input[i..].chars().next().unwrap();
131        out.push(ch);
132        i += ch.len_utf8();
133    }
134
135    out
136}

138/// Maps select structural HTML tags to clean Markdown equivalents; drops all others.
139fn map_structural_tag(raw_tag: &str) -> &'static str {
140    let inner = raw_tag
141        .trim_start_matches('<')
142        .trim_end_matches('>')
143        .trim();
144
145    let tag_name = inner
146        .trim_start_matches('/')
147        .split(|c: char| c.is_whitespace() || c == '/')
148        .next()
149        .unwrap_or("")
150        .to_ascii_lowercase();
151
152    let is_closing = inner.starts_with('/');
153
154    match tag_name.as_str() {
155        "br" => "  \n",
156        "hr" => "\n---\n",
157        "summary" if !is_closing => "**",
158        "summary" if is_closing => "**\n",
159        _ => "",
160    }
161}

163/// Rewrites `[label](target)` Markdown links to clean relative `.md` paths for Shortcut Docs.
164fn rewrite_markdown_links(input: &str) -> String {
165    let mut out = String::with_capacity(input.len());
166    let mut rest = input;
167
168    while let Some(open_bracket) = rest.find('[') {
169        out.push_str(&rest[..open_bracket]);
170        let after_open = &rest[open_bracket..];
171
172        if let Some(mid) = after_open.find("](") {
173            let label = &after_open[1..mid];
174            let after_mid = &after_open[mid + 2..];
175
176            if let Some(close_paren) = after_mid.find(')') {
177                let raw_target = after_mid[..close_paren].trim();
178                let clean_target = normalize_target_path(raw_target);
179
180                out.push('[');
181                out.push_str(label);
182                out.push_str("](");
183                out.push_str(&clean_target);
184                out.push(')');
185
186                rest = &after_mid[close_paren + 1..];
187                continue;
188            }
189        }
190
191        out.push('[');
192        rest = &after_open[1..];
193    }
194
195    out.push_str(rest);
196    out
197}

199/// Normalizes internal doc targets into relative `.md` paths while preserving external URLs.
200fn normalize_target_path(target: &str) -> String {
201    if target.starts_with("http://")
202        || target.starts_with("https://")
203        || target.starts_with("mailto:")
204    {
205        return target.to_string();
206    }
207
208    if let Some(anchor) = target.strip_prefix('#') {
209        return format!("#{}", slugify_anchor(anchor));
210    }
211
212    let (raw_path, fragment) = match target.split_once('#') {
213        Some((p, f)) => (p, Some(f)),
214        None => (target, None),
215    };
216
217    let stripped_prefix = strip_version_prefix(raw_path.trim_start_matches('/'));
218
219    let relative_md = if stripped_prefix.ends_with("/index.html") {
220        let dir = stripped_prefix.trim_end_matches("/index.html");
221        format!("./{dir}.md")
222    } else if stripped_prefix == "index.html" {
223        "./index.md".to_string()
224    } else if let Some(stem) = stripped_prefix.strip_suffix(".html") {
225        if stem.starts_with("./") || stem.starts_with("../") {
226            format!("{stem}.md")
227        } else {
228            format!("./{stem}.md")
229        }
230    } else if stripped_prefix.ends_with(".md") {
231        if stripped_prefix.starts_with("./") || stripped_prefix.starts_with("../") {
232            stripped_prefix.to_string()
233        } else {
234            format!("./{stripped_prefix}")
235        }
236    } else {
237        stripped_prefix.to_string()
238    };
239
240    match fragment {
241        Some(f) if !f.is_empty() => format!("{relative_md}#{}", slugify_anchor(f)),
242        _ => relative_md,
243    }
244}

246fn strip_version_prefix(path: &str) -> &str {
247    if let Some(rest) = path.strip_prefix("latest/") {
248        return rest;
249    }
250    if let Some(rest) = path.strip_prefix('v') {
251        if let Some((ver, remainder)) = rest.split_once('/') {
252            let is_semver = ver
253                .split('.')
254                .all(|part| !part.is_empty() && part.chars().all(|c| c.is_ascii_digit()));
255            if is_semver {
256                return remainder;
257            }
258        }
259    }
260    path
261}

263fn slugify_anchor(anchor: &str) -> String {
264    anchor
265        .trim()
266        .to_ascii_lowercase()
267        .chars()
268        .map(|c| {
269            if c.is_ascii_alphanumeric() || c == '-' || c == '_' {
270                c
271            } else {
272                '-'
273            }
274        })
275        .collect::<String>()
276        .trim_matches('-')
277        .to_string()
278}

280/// Prevents more than two consecutive newlines after stripping block-level HTML wrappers.
281fn collapse_excess_blank_lines(input: &str) -> String {
282    let mut out = String::with_capacity(input.len());
283    let mut consecutive_newlines = 0;
284
285    for ch in input.chars() {
286        if ch == '\n' {
287            consecutive_newlines += 1;
288            if consecutive_newlines <= 2 {
289                out.push(ch);
290            }
291        } else {
292            consecutive_newlines = 0;
293            out.push(ch);
294        }
295    }
296
297    out
298}

304    #[test]
305    fn strips_raw_html_outside_code_fences_only() {
306        let input = concat!(
307            "# `JournalBatch`\n\n",
308            "<div class=\"topbar\">Version v0.1.0</div>\n",
309            "<details><summary>Invariants</summary>\n",
310            "Zero-sum check<br/>Must balance.\n",
311            "</details>\n\n",
312            "Inline code keeps `Result<T, E>` and `<span>raw</span>` untouched.\n\n",
313            "~~~\n",
314            "fn verify<T: Ord>(items: Vec<T>) -> Result<(), String> {\n",
315            "    println!(\"<div>not stripped</div>\");\n",
316            "    Ok(())\n",
317            "}\n",
318            "~~~\n"
319        );
320
321        let cleaned = sanitize_for_shortcut_docs(input);
322
323        assert!(!cleaned.contains("<div class=\"topbar\">"));
324        assert!(!cleaned.contains("<details>"));
325        assert!(!cleaned.contains("<br/>"));
326        assert!(cleaned.contains("**Invariants**\n"));
327        assert!(cleaned.contains("Zero-sum check  \nMust balance."));
328        assert!(cleaned.contains("`Result<T, E>`"));
329        assert!(cleaned.contains("`<span>raw</span>`"));
330        assert!(cleaned.contains("println!(\"<div>not stripped</div>\");"));
331    }

333    #[test]
334    fn rewrites_rustdoc_and_versioned_links_to_relative_md() {
335        let input = concat!(
336            "See [Tuos Ledger](/latest/tuos/index.html) and ",
337            "[JournalBatch](/v0.1.0/tuos/struct.JournalBatch.html#method.commit).\n",
338            "External [GitHub](https://github.com/Hara-Horizon/hh-docversion) stays intact.\n",
339            "Local anchor [Invariants](#System Invariants) is slugified.\n"
340        );
341
342        let cleaned = sanitize_for_shortcut_docs(input);
343
344        assert!(cleaned.contains("[Tuos Ledger](./tuos.md)"));
345        assert!(cleaned.contains("[JournalBatch](./tuos/struct.JournalBatch.md#method-commit)"));
346        assert!(cleaned.contains("[GitHub](https://github.com/Hara-Horizon/hh-docversion)"));
347        assert!(cleaned.contains("[Invariants](#system-invariants)"));
348    }
